# Logic/Purpose: Compare non-key attributes row-by-row on matching keys. 
# Data represented: For rows where keys match, count mismatches for each listed column. 
 
from pyspark.sql import functions as F 
 
keys = ["LOT", "DATE_TIME", "HISTORDER"] 
 
cols_to_compare = [ 
    "OPER", "OPERDESC", "OPERLONGDESC", "TRANS", 
    "HIST_REC", "MACHINE", "USERNAME", "COMMAND", "EMPID" 
] 
 
compare = df_Prod.alias("p").join(df_Fabric.alias("f"), keys) 
 
result = compare.agg(*[ 
    F.sum( 
        F.when( 
            F.coalesce(F.col(f"p.{c}"), F.lit("")) != 
            F.coalesce(F.col(f"f.{c}"), F.lit("")), 
            1 
        ).otherwise(0) 
    ).alias(f"{c}_MISMATCH") 
    for c in cols_to_compare 
]) 
 
result.show(truncate=False) 
 

 
 

Result 
Single-row mismatch summary, e.g.: OPER_MISMATCH, TRANS_MISMATCH, HIST_REC_MISMATCH, etc., with counts of mismatching rows.
