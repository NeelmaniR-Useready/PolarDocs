%%pyspark

from pyspark.sql import functions as F

# ============================================================
# COLUMN NAMES
# ============================================================

column_names = [
    "DATE_TIME_RUN",
    "EMPID",
    "MACHINE",
    "MACHINE_DESCRIPTION",
    "MACHINE_GROUP",
    "MACHINE_TYPE",
    "MACHINE_KIND",
    "DATE_TIME",
    "STATUS1_CODE",
    "STATUS1_NAME",
    "STATUS2_CODE",
    "STATUS2_NAME",
    "PM_CODE",
    "PM_NAME",
    "REPAIR1_CODE",
    "REPAIR1_NAME",
    "REPAIR2_CODE",
    "REPAIR2_NAME",
    "REPAIR3_CODE",
    "REPAIR3_NAME",
    "IGNORE_RECORD",
    "COMMENTS",
    "USERNAME",
    "COMMENTTYPE",
    "LINEORDER",
    "MACHINE_PRIORITY",
    "REPAIRCODE"
]

# ============================================================
# LOAD DATA (NO HEADER)
# ============================================================

df_Prod = spark.read.csv(
    "abfss://9dd39a9d-2357-46a3-913d-1ff996531e62@onelake.dfs.fabric.microsoft.com/26a167a1-535e-4366-95da-c08c7730c83b/Files/Production1jan1yearv.csv",
    header=False,
    inferSchema=True
)

df_Fabric = spark.read.csv(
    "abfss://9dd39a9d-2357-46a3-913d-1ff996531e62@onelake.dfs.fabric.microsoft.com/26a167a1-535e-4366-95da-c08c7730c83b/Files/Fabric1jan1yearv.csv",
    header=False,
    inferSchema=True
)

# ============================================================
# ASSIGN COLUMN NAMES
# ============================================================

df_Prod = df_Prod.toDF(*column_names)
df_Fabric = df_Fabric.toDF(*column_names)

# ============================================================
# OPTIONAL CLEANUP
# ============================================================

def trim_all(df):
    return df.select([
        F.trim(F.col(c)).alias(c)
        if df.schema[c].dataType.simpleString() == "string"
        else F.col(c)
        for c in df.columns
    ])

df_Prod = trim_all(df_Prod)
df_Fabric = trim_all(df_Fabric)

# ============================================================
# CACHE
# ============================================================

df_Prod.cache()
df_Fabric.cache()

all_cols = column_names

# ============================================================
# ROW COUNTS
# ============================================================

print("=" * 100)
print("ROW COUNTS")
print("=" * 100)

prod_count = df_Prod.count()
fabric_count = df_Fabric.count()

print("Prod Rows  :", prod_count)
print("Fabric Rows:", fabric_count)

# ============================================================
# DISTINCT COUNTS
# ============================================================

print("=" * 100)
print("PROD DISTINCT COUNTS")
print("=" * 100)

df_Prod.select(
    *[F.countDistinct(c).alias(c) for c in all_cols]
).show(vertical=True, truncate=False)

print("=" * 100)
print("FABRIC DISTINCT COUNTS")
print("=" * 100)

df_Fabric.select(
    *[F.countDistinct(c).alias(c) for c in all_cols]
).show(vertical=True, truncate=False)

# ============================================================
# NULL / EMPTY PROFILE
# ============================================================

print("=" * 100)
print("PROD NULL PROFILE")
print("=" * 100)

df_Prod.select([
    F.sum(
        F.when(
            F.col(c).isNull() |
            (F.trim(F.col(c).cast("string")) == ""),
            1
        ).otherwise(0)
    ).alias(c)
    for c in all_cols
]).show(vertical=True)

print("=" * 100)
print("FABRIC NULL PROFILE")
print("=" * 100)

df_Fabric.select([
    F.sum(
        F.when(
            F.col(c).isNull() |
            (F.trim(F.col(c).cast("string")) == ""),
            1
        ).otherwise(0)
    ).alias(c)
    for c in all_cols
]).show(vertical=True)

# ============================================================
# DUPLICATES
# ============================================================

print("=" * 100)
print("DUPLICATE ANALYSIS")
print("=" * 100)

print(
    "Prod Duplicate Rows:",
    prod_count - df_Prod.distinct().count()
)

print(
    "Fabric Duplicate Rows:",
    fabric_count - df_Fabric.distinct().count()
)

# ============================================================
# FULL ROW COMPARISON
# ============================================================

print("=" * 100)
print("FULL ROW COMPARISON")
print("=" * 100)

common_rows = df_Prod.intersect(df_Fabric).count()

only_prod = df_Prod.exceptAll(df_Fabric)
only_fabric = df_Fabric.exceptAll(df_Prod)

only_prod_count = only_prod.count()
only_fabric_count = only_fabric.count()

print("Common Rows    :", common_rows)
print("Only In Prod   :", only_prod_count)
print("Only In Fabric :", only_fabric_count)

# ============================================================
# DATE RANGE ANALYSIS
# ============================================================

for c in ["DATE_TIME_RUN", "DATE_TIME"]:

    print("=" * 100)
    print(f"DATE ANALYSIS : {c}")
    print("=" * 100)

    print("PROD")

    df_Prod.select(
        F.min(c).alias("MIN"),
        F.max(c).alias("MAX")
    ).show(truncate=False)

    print("FABRIC")

    df_Fabric.select(
        F.min(c).alias("MIN"),
        F.max(c).alias("MAX")
    ).show(truncate=False)

# ============================================================
# KEY COLUMNS
# ============================================================

key_cols = [
    "MACHINE",
    "DATE_TIME",
    "LINEORDER"
]

print("Keys:", key_cols)

# ============================================================
# ATTRIBUTE MISMATCH ANALYSIS
# ============================================================

compare = (
    df_Prod.alias("p")
    .join(
        df_Fabric.alias("f"),
        key_cols,
        "inner"
    )
)

compare_cols = [
    c for c in all_cols
    if c not in key_cols
]

mismatch_summary = compare.agg(*[
    F.sum(
        F.when(
            F.coalesce(
                F.col(f"p.{c}").cast("string"),
                F.lit("")
            )
            !=
            F.coalesce(
                F.col(f"f.{c}").cast("string"),
                F.lit("")
            ),
            1
        ).otherwise(0)
    ).alias(f"{c}_MISMATCH")
    for c in compare_cols
])

mismatch_summary.show(
    vertical=True,
    truncate=False
)

# ============================================================
# ROWS ONLY IN PROD
# ============================================================

print("=" * 100)
print("ROWS ONLY IN PROD")
print("=" * 100)

only_prod.show(
    100,
    truncate=False
)

# ============================================================
# ROWS ONLY IN FABRIC
# ============================================================

print("=" * 100)
print("ROWS ONLY IN FABRIC")
print("=" * 100)

only_fabric.show(
    100,
    truncate=False
)
