%%pyspark

from pyspark.sql import functions as F

# ============================================================
# LOAD DATA
# ============================================================

df_Prod = spark.read.csv(
    "abfss://9dd39a9d-2357-46a3-913d-1ff996531e62@onelake.dfs.fabric.microsoft.com/26a167a1-535e-4366-95da-c08c7730c83b/Files/Production1march1week.csv",
    header=True,
    inferSchema=True
)

df_Fabric = spark.read.csv(
    "abfss://9dd39a9d-2357-46a3-913d-1ff996531e62@onelake.dfs.fabric.microsoft.com/26a167a1-535e-4366-95da-c08c7730c83b/Files/Fabric1march1week.csv",
    header=True,
    inferSchema=True
)

df_Prod.cache()
df_Fabric.cache()

all_cols = df_Prod.columns

# ============================================================
# ROW COUNTS
# ============================================================

print("\n" + "=" * 100)
print("ROW COUNTS")
print("=" * 100)

prod_count = df_Prod.count()
fabric_count = df_Fabric.count()

print("Prod Rows   :", prod_count)
print("Fabric Rows :", fabric_count)

# ============================================================
# DISTINCT PROFILE
# ============================================================

print("\n" + "=" * 100)
print("PROD DISTINCT COUNTS")
print("=" * 100)

df_Prod.select(
    *[
        F.countDistinct(c).alias(c)
        for c in all_cols
    ]
).show(vertical=True, truncate=False)

print("\n" + "=" * 100)
print("FABRIC DISTINCT COUNTS")
print("=" * 100)

df_Fabric.select(
    *[
        F.countDistinct(c).alias(c)
        for c in all_cols
    ]
).show(vertical=True, truncate=False)

# ============================================================
# NULL PROFILE
# ============================================================

print("\n" + "=" * 100)
print("PROD NULL PROFILE")
print("=" * 100)

df_Prod.select([
    F.sum(
        F.when(
            F.col(c).isNull()
            | (F.trim(F.col(c).cast("string")) == ""),
            1
        ).otherwise(0)
    ).alias(c)
    for c in all_cols
]).show(vertical=True)

print("\n" + "=" * 100)
print("FABRIC NULL PROFILE")
print("=" * 100)

df_Fabric.select([
    F.sum(
        F.when(
            F.col(c).isNull()
            | (F.trim(F.col(c).cast("string")) == ""),
            1
        ).otherwise(0)
    ).alias(c)
    for c in all_cols
]).show(vertical=True)

# ============================================================
# DUPLICATE ANALYSIS
# ============================================================

print("\n" + "=" * 100)
print("DUPLICATE ANALYSIS")
print("=" * 100)

prod_distinct = df_Prod.distinct().count()
fabric_distinct = df_Fabric.distinct().count()

print("Prod Duplicate Rows   :", prod_count - prod_distinct)
print("Fabric Duplicate Rows :", fabric_count - fabric_distinct)

# ============================================================
# FULL ROW COMPARISON
# ============================================================

print("\n" + "=" * 100)
print("FULL ROW COMPARISON")
print("=" * 100)

common_rows = df_Prod.intersect(df_Fabric).count()

only_prod = df_Prod.exceptAll(df_Fabric)
only_fabric = df_Fabric.exceptAll(df_Prod)

only_prod_count = only_prod.count()
only_fabric_count = only_fabric.count()

print("Common Rows      :", common_rows)
print("Only In Prod     :", only_prod_count)
print("Only In Fabric   :", only_fabric_count)

if prod_count > 0:
    print(
        "Prod Match %     :",
        round(common_rows * 100 / prod_count, 2)
    )

if fabric_count > 0:
    print(
        "Fabric Match %   :",
        round(common_rows * 100 / fabric_count, 2)
    )

# ============================================================
# DATE RANGE ANALYSIS
# ============================================================

date_cols = [
    c for c in all_cols
    if "DATE" in c.upper()
]

for c in date_cols:

    print("\n" + "=" * 100)
    print(f"DATE RANGE ANALYSIS : {c}")
    print("=" * 100)

    print("PROD")

    df_Prod.select(
        F.min(c).alias("MIN_VALUE"),
        F.max(c).alias("MAX_VALUE")
    ).show(truncate=False)

    print("FABRIC")

    df_Fabric.select(
        F.min(c).alias("MIN_VALUE"),
        F.max(c).alias("MAX_VALUE")
    ).show(truncate=False)

# ============================================================
# DISTRIBUTION ANALYSIS FOR EVERY COLUMN
# ============================================================

for c in all_cols:

    print("\n" + "=" * 100)
    print(f"DISTRIBUTION ANALYSIS : {c}")
    print("=" * 100)

    prod_dist = (
        df_Prod.groupBy(c)
        .count()
        .withColumnRenamed("count", "prod_count")
    )

    fabric_dist = (
        df_Fabric.groupBy(c)
        .count()
        .withColumnRenamed("count", "fabric_count")
    )

    compare_dist = (
        prod_dist.join(
            fabric_dist,
            c,
            "full"
        )
        .fillna(0)
        .withColumn(
            "difference",
            F.col("fabric_count") - F.col("prod_count")
        )
    )

    compare_dist.orderBy(
        F.desc(F.abs(F.col("difference")))
    ).limit(20).show(20, False)

# ============================================================
# SUMMARY METRICS
# ============================================================

print("\n" + "=" * 100)
print("SUMMARY METRICS")
print("=" * 100)

summary_prod = df_Prod.select(
    F.count("*").alias("rows")
)

summary_fabric = df_Fabric.select(
    F.count("*").alias("rows")
)

print("PROD")
summary_prod.show(truncate=False)

print("FABRIC")
summary_fabric.show(truncate=False)

# ============================================================
# KEY COLUMNS
# ============================================================

key_cols = [
    "MACHINE",
    "DATE_TIME",
    "LINEORDER"
]

key_cols = [
    c for c in key_cols
    if c in all_cols
]

print("\nDetected Keys :", key_cols)

# ============================================================
# ROW LEVEL ATTRIBUTE COMPARISON
# ============================================================

if len(key_cols) > 0:

    print("\n" + "=" * 100)
    print("ATTRIBUTE MISMATCH ANALYSIS")
    print("=" * 100)

    compare = (
        df_Prod.alias("p")
        .join(
            df_Fabric.alias("f"),
            key_cols,
            "inner"
        )
    )

    compare_cols = [
        c
        for c in all_cols
        if c not in key_cols
    ]

    mismatch_summary = compare.agg(*[
        F.sum(
            F.when(
                F.coalesce(
                    F.col(f"p.{c}").cast("string"),
                    F.lit("")
                )
                !=
                F.coalesce(
                    F.col(f"f.{c}").cast("string"),
                    F.lit("")
                ),
                1
            ).otherwise(0)
        ).alias(f"{c}_MISMATCH")
        for c in compare_cols
    ])

    mismatch_summary.show(
        vertical=True,
        truncate=False
    )

# ============================================================
# COLUMN-BY-COLUMN MISMATCH DETAILS
# ============================================================

    for c in compare_cols:

        mismatch_df = compare.filter(
            F.coalesce(
                F.col(f"p.{c}").cast("string"),
                F.lit("")
            )
            !=
            F.coalesce(
                F.col(f"f.{c}").cast("string"),
                F.lit("")
            )
        )

        if mismatch_df.limit(1).count() > 0:

            print("\n" + "=" * 100)
            print(f"MISMATCH SAMPLE : {c}")
            print("=" * 100)

            mismatch_df.select(
                *key_cols,
                F.col(f"p.{c}").alias(f"PROD_{c}"),
                F.col(f"f.{c}").alias(f"FABRIC_{c}")
            ).limit(50).show(
                50,
                False
            )

# ============================================================
# ROWS ONLY IN PROD
# ============================================================

print("\n" + "=" * 100)
print("ROWS ONLY IN PROD")
print("=" * 100)

only_prod.show(
    50,
    False
)

# ============================================================
# ROWS ONLY IN FABRIC
# ============================================================

print("\n" + "=" * 100)
print("ROWS ONLY IN FABRIC")
print("=" * 100)

only_fabric.show(
    50,
    False
)
